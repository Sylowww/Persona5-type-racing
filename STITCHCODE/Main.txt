<script>
    // Desktop keyboard shortcuts: Enter / Space triggers primary start
    document.addEventListener('keydown', (e) => {
      const activeTag = document.activeElement ? document.activeElement.tagName.toLowerCase() : '';
      if (activeTag === 'input' || activeTag === 'textarea') return;

      if (e.code === 'Enter' || e.code === 'Space') {
        e.preventDefault();
        const mainBtn = document.getElementById('btn-engage-main');
        if (mainBtn) {
          mainBtn.classList.add('scale-95');
          setTimeout(() => {
            mainBtn.classList.remove('scale-95');
            // Simulating prompt engagement feedback
            const originalColor = mainBtn.querySelector('.bg-primary-container');
            if (originalColor) {
              originalColor.classList.add('bg-secondary-container');
              setTimeout(() => originalColor.classList.remove('bg-secondary-container'), 250);
            }
          }, 120);
        }
      }
    });

    // Mechanical switch sound toggle behavior
    const btnClicky = document.getElementById('btn-audio-clicky');
    const btnLinear = document.getElementById('btn-audio-linear');
    if (btnClicky && btnLinear) {
      btnClicky.addEventListener('click', () => {
        btnClicky.className = 'px-2 py-0.5 font-label-hud text-[10px] uppercase font-black bg-primary-container text-on-primary-container transition-all';
        btnLinear.className = 'px-2 py-0.5 font-label-hud text-[10px] uppercase font-black text-on-surface-variant hover:text-secondary transition-all';
      });
      btnLinear.addEventListener('click', () => {
        btnLinear.className = 'px-2 py-0.5 font-label-hud text-[10px] uppercase font-black bg-secondary-container text-on-secondary-fixed transition-all';
        btnClicky.className = 'px-2 py-0.5 font-label-hud text-[10px] uppercase font-black text-on-surface-variant hover:text-secondary transition-all';
      });
    }
  </script>
